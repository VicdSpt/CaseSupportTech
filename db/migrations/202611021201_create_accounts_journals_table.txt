CREATE TABLE IF NOT EXISTS accounts_journals(
    `id` INT(11) NOT NULL AUTO_INCREMENT,
    `account_id` INT(11) NOT NULL,
    `journal_id` INT(11) NOT NULL,
    `created_at` datetime NOT NULL,
    `updated_at` datetime NOT NULL,
    PRIMARY KEY (`id`),
    CONSTRAINT unique_journal_per_account UNIQUE (`account_id`, `journal_id`),
    CONSTRAINT `accounts_journals_ibfk_1` FOREIGN KEY (`account_id`) REFERENCES `accounts` (`id`),
    CONSTRAINT `accounts_journals_ibfk_2` FOREIGN KEY (`journal_id`) REFERENCES `journals` (`id`)
);
