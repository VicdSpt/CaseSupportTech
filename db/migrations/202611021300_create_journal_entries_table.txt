CREATE TABLE IF NOT EXISTS journal_entries(
    `id` INT(11) NOT NULL AUTO_INCREMENT,
    `db` VARCHAR(30) NOT NULL,
    `move_id` INT(11) NOT NULL,
    `journal_id` INT(11) NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `ref` TEXT DEFAULT NULL,
    `date` DATE NOT NULL,
    `state` VARCHAR(20) NOT NULL,
    `move_type` VARCHAR(20) NOT NULL,
    `partner_odoo_id` INT(11) DEFAULT NULL,
    `currency` VARCHAR(10) DEFAULT NULL,
    `amount_total` DECIMAL(20, 4) NOT NULL DEFAULT 0,
    `odoo_write_date` datetime NOT NULL,
    `created_at` datetime NOT NULL,
    `updated_at` datetime NOT NULL,
    PRIMARY KEY (`id`),
    CONSTRAINT unique_move_id_per_db UNIQUE (`db`, `move_id`),
    INDEX journal_entries_journal_date (`journal_id`, `date`),
    CONSTRAINT `journal_entries_ibfk_1` FOREIGN KEY (`journal_id`) REFERENCES `journals` (`id`)
);
